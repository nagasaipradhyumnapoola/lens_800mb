# whisper.cpp built as shared libraries with runtime-selected ggml CPU backends (GGML_BACKEND_DL +
# GGML_CPU_ALL_VARIANTS): no binary requires AVX/AVX2 unconditionally (docs/LENS_V1_SPEC.md §15).
set(WHISPER_BUILD "${LENS_ROOT}/build/whisper-dl")
set(WHISPER_LIBS "${WHISPER_BUILD}/src/Release/whisper.lib" "${WHISPER_BUILD}/examples/Release/common.lib"
  "${WHISPER_BUILD}/ggml/src/Release/ggml.lib" "${WHISPER_BUILD}/ggml/src/Release/ggml-base.lib")
# Delay-loaded so the engine starts, and idles, without mapping ASR code until audio is transcribed.
set(WHISPER_DELAYLOAD /DELAYLOAD:whisper.dll /DELAYLOAD:ggml.dll /DELAYLOAD:ggml-base.dll)
file(GLOB WHISPER_RUNTIME "${WHISPER_BUILD}/bin/Release/whisper.dll" "${WHISPER_BUILD}/bin/Release/ggml.dll"
  "${WHISPER_BUILD}/bin/Release/ggml-base.dll" "${WHISPER_BUILD}/bin/Release/ggml-cpu-*.dll")
function(lens_use_whisper target)
  target_link_libraries(${target} PRIVATE ${WHISPER_LIBS} delayimp)
  target_link_options(${target} PRIVATE ${WHISPER_DELAYLOAD})
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${WHISPER_RUNTIME} $<TARGET_FILE_DIR:${target}>)
endfunction()
add_executable(lens-asr-probe asr_probe.cpp platform.cpp)
target_include_directories(lens-asr-probe PRIVATE "${LENS_ROOT}/third_party/whisper.cpp/include"
  "${LENS_ROOT}/third_party/whisper.cpp/examples" "${LENS_ROOT}/third_party/whisper.cpp/ggml/include")
target_link_libraries(lens-asr-probe PRIVATE psapi)
lens_use_whisper(lens-asr-probe)
target_compile_options(lens-asr-probe PRIVATE /utf-8 /W4 /sdl)
